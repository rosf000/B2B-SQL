def calculate_achievement(
    actual_revenue: float,
    target: float
) -> dict:
    """計算業績達成率，並回傳分析結果。"""
    if target <= 0:
        raise ValueError(f"目標業績必須大於 0，收到：{target}")

    return {

        "target": target,
        "rate_pct": round(rate, 1),
        "status": "達標" if rate >= 100 else "未達標"
    }

### 1.3 Decorator（裝飾器）

裝飾器是 Python 最優雅的特性之一，讓你在不修改函式本身的情況下，添加額外的功能（如計時、重試、驗證）。

**基本概念**：裝飾器是一個接收函式的函式，回傳一個新的函式。

```python
import time
import functools

# 裝飾器範例 1：計時器（測量函式執行時間）
def timer(func):
    @functools.wraps(func)   # 保留原函式的 __name__ 和 __doc__
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)       # 執行原函式
        elapsed = time.perf_counter() - start
        print(f"[TIMER] {func.__name__} 執行時間：{elapsed:.3f}s")
        return result
    return wrapper

@timer
def slow_query():
    """模擬一個慢查詢。"""
    time.sleep(1.5)
    return "查詢完成"

slow_query()  # 自動列印執行時間
# [TIMER] slow_query 執行時間：1.502s
```

```python
# 裝飾器範例 2：重試機制（網路請求或資料庫連線失敗時自動重試）
import time

def retry(max_attempts: int = 3, delay: float = 1.0, exceptions=(Exception,)):
    """
    自動重試裝飾器。

    Args:
        max_attempts: 最大重試次數
        delay: 每次重試前等待的秒數
        exceptions: 要捕捉的例外類型
    """
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(1, max_attempts + 1):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    if attempt < max_attempts:
                        print(f"[RETRY] {func.__name__} 第 {attempt} 次失敗：{e}，{delay}s 後重試...")
                        time.sleep(delay)
                    else:
                        print(f"[RETRY] {func.__name__} 已達最大重試次數 ({max_attempts})")
            raise last_exception
        return wrapper
    return decorator

@retry(max_attempts=3, delay=2.0, exceptions=(ConnectionError,))
def fetch_from_api(url: str) -> dict:
    """從 API 取得資料（可能因網路問題失敗）。"""
    # 實際程式碼會呼叫 requests.get(url)
    raise ConnectionError("連線逾時")
```

```python
# 裝飾器範例 3：輸入驗證
def validate_positive(func):
    """確保函式的所有數值引數都是正數。"""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        for arg in args:
            if isinstance(arg, (int, float)) and arg <= 0:
                raise ValueError(f"{func.__name__} 的引數必須是正數，收到：{arg}")
        return func(*args, **kwargs)
    return wrapper

@validate_positive
def calculate_margin(revenue: float, cost: float) -> float:
    return (revenue - cost) / revenue * 100

calculate_margin(100_000, 60_000)   # ✅ 正常
calculate_margin(-100_000, 60_000)  # ❌ 拋出 ValueError
```

## 三、例外處理（Exception Handling）

### 3.1 try / except / else / finally 完整語法

```python
def safe_divide(a: float, b: float) -> float:
    """安全的除法，處理除以零的情況。"""
    try:
        result = a / b          # 可能出錯的程式碼
    except ZeroDivisionError:
        print("錯誤：除數不能為零")
        return 0.0
    except TypeError as e:
        print(f"型別錯誤：{e}")
        raise                   # 重新拋出，讓呼叫者處理
    else:
        # 只在 try 成功（沒有例外）時執行
        print(f"計算成功：{a} / {b} = {result}")
        return result
    finally:
        # 無論成功或失敗都會執行（適合釋放資源）
        print("除法操作結束")
```

# 呼叫端的處理

## 四、logging 模組 — 取代 print 的最佳實踐

### 4.1 為什麼不要用 print？

| | `print` | `logging` |
|---|---|---|
| 日誌等級 | 無 | DEBUG/INFO/WARNING/ERROR/CRITICAL |
| 時間戳記 | 需手動加 | 自動加 |
| 輸出目標 | 只能螢幕 | 螢幕 + 檔案 + 遠端 |
| 生產環境關閉 | 要手動刪除 | 調整 level 即可 |
| 格式化 | 手動 | 統一設定 |

### 4.2 logging 完整設定

```python
# config/logging_config.py
import logging
import logging.handlers
import os
from datetime import datetime

def setup_logging(
    log_name: str = "b2b_app",
    log_dir: str = "logs",
    level: int = logging.INFO
) -> logging.Logger:
    """
    設定應用程式的日誌系統。

    Returns:
        設定好的 Logger 物件
    """
    # 建立 logs 資料夾
    os.makedirs(log_dir, exist_ok=True)

    # 建立 Logger
    logger = logging.getLogger(log_name)
    logger.setLevel(logging.DEBUG)  # Logger 本身設最低，由 Handler 控制輸出等級

    # 格式化字串
    formatter = logging.Formatter(
        fmt="%(asctime)s [%(levelname)-8s] %(name)s:%(lineno)d - %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S"
    )
    # 輸出範例：
    # 2024-09-15 10:30:45 [INFO    ] b2b_app:42 - 成功建立訂單 ORD-2024-001

    # Handler 1：輸出到終端（DEBUG 以上）
    console_handler = logging.StreamHandler()
    console_handler.setLevel(logging.DEBUG)
    console_handler.setFormatter(formatter)

    # Handler 2：輸出到檔案，自動按日期切割（INFO 以上）
    log_file = os.path.join(log_dir, f"{log_name}.log")
    file_handler = logging.handlers.TimedRotatingFileHandler(
        filename=log_file,
        when="midnight",    # 每天午夜切割
        backupCount=30,     # 保留 30 天的日誌
        encoding="utf-8"
    )
    file_handler.setLevel(logging.INFO)
    file_handler.setFormatter(formatter)

    # Handler 3：只記錄 ERROR 以上的嚴重錯誤（可串接到告警系統）
    error_log_file = os.path.join(log_dir, f"{log_name}_error.log")
    error_handler = logging.FileHandler(error_log_file, encoding="utf-8")
    error_handler.setLevel(logging.ERROR)
    error_handler.setFormatter(formatter)

    # 加入所有 Handler（避免重複加入）
    if not logger.handlers:
        logger.addHandler(console_handler)
        logger.addHandler(file_handler)
        logger.addHandler(error_handler)

    return logger


# 使用範例
logger = setup_logging("etl_pipeline")

def process_orders_etl(date_str: str):
    logger.info(f"開始處理 {date_str} 的訂單 ETL")

    try:
        orders = fetch_orders(date_str)
        logger.debug(f"取得 {len(orders)} 筆訂單")

        for order in orders:
            try:
                process_single_order(order)
                logger.info(f"訂單 {order['order_number']} 處理完成")
            except InsufficientStockError as e:
                logger.warning(f"庫存不足跳過：{e}")
            except Exception as e:
                logger.error(f"訂單 {order['order_number']} 處理失敗：{e}", exc_info=True)
                # exc_info=True 會附上完整的 traceback

    except Exception as e:
        logger.critical(f"ETL 管線崩潰：{e}", exc_info=True)
        raise

    logger.info(f"{date_str} ETL 完成")
```

### 5.1 pdb — Python 內建除錯器

```python
# 方式 1：直接在程式碼中插入中斷點
import pdb

def buggy_function(orders):
    total = 0
    for order in orders:
        pdb.set_trace()    # 程式在此暫停，進入互動式除錯
        total += order["amount"]  # 可以在這裡檢查 order 的內容
    return total

# pdb 常用指令：
# n (next)      → 執行下一行
# s (step)      → 進入函式內部
# c (continue)  → 繼續執行到下一個中斷點
# p <expr>      → 列印運算式的值（如 p order["amount"]）
# pp <expr>     → 漂亮列印（pretty print）
# l (list)      → 顯示當前位置附近的程式碼
# q (quit)      → 退出除錯器

# 方式 2：Python 3.7+ 的 breakpoint()（不需要 import pdb）
def process_data(data):
    result = []
    for item in data:
        breakpoint()    # 等同於 pdb.set_trace()
        processed = transform(item)
        result.append(processed)
    return result
```

## 六、實用模式：Context Manager（with 語句）

```python
# 資料庫連線的正確關閉方式
# ❌ 危險寫法：若中途出錯，連線永遠不會關閉！
conn = db.connect()
result = conn.execute("SELECT ...")  # 若這裡出錯...
conn.close()  # ...這行永遠不會執行

# ✅ 正確寫法：with 確保無論如何都會執行 close
with db.connect() as conn:
    result = conn.execute("SELECT ...")
# 離開 with 區塊時，自動呼叫 conn.close()

# 讀寫檔案的標準寫法
with open("orders.csv", "r", encoding="utf-8") as f:
    content = f.read()
# 不需要 f.close()，with 自動處理

# 自定義 Context Manager（用 dataclass 或 contextlib）
from contextlib import contextmanager

@contextmanager
def db_transaction(conn):
    """資料庫交易的 Context Manager。"""
    try:
        yield conn              # 進入 with 區塊
        conn.commit()           # 成功：提交
    except Exception:
        conn.rollback()         # 失敗：回滾
        raise
    finally:
        conn.close()            # 無論如何：關閉連線

# 使用
with db_transaction(get_connection()) as conn:
    conn.execute("UPDATE orders SET status='COMPLETED' WHERE order_id=1")
    conn.execute("UPDATE products SET stock=stock-2 WHERE product_id=101")
# 兩個 UPDATE 都成功 → 自動 commit
# 任一 UPDATE 失敗 → 自動 rollback
```

```python
# src/utils.py — 專案通用工具函式

import logging
import os
from typing import Any, Optional
from dotenv import load_dotenv

load_dotenv()  # 從 .env 檔案讀取環境變數

logger = logging.getLogger(__name__)


def get_env(key: str, default: Optional[str] = None) -> str:
    """
    安全地取得環境變數。

    Args:
        key: 環境變數名稱
        default: 若不存在時的預設值（若無預設且不存在，拋出例外）

    Raises:
        EnvironmentError: 環境變數不存在且沒有預設值
    """
    value = os.getenv(key, default)
    if value is None:
        raise EnvironmentError(f"必要的環境變數 '{key}' 未設定。請在 .env 檔案中加入此變數。")
    return value


def format_currency(amount: float, symbol: str = "NT$") -> str:
    """將數值格式化為貨幣字串。"""
    return f"{symbol}{amount:,.0f}"


def chunks(lst: list, n: int):
    """
    將清單切分為大小為 n 的批次（Generator）。

    用法：
        for batch in chunks(order_ids, 100):
            process_batch(batch)
    """
    for i in range(0, len(lst), n):
        yield lst[i:i + n]


def safe_get(data: dict, *keys: str, default: Any = None) -> Any:
    """
    安全地從巢狀字典取值，避免 KeyError。

    用法：
        safe_get(order, "customer", "city")  # order["customer"]["city"]
        safe_get(order, "customer", "phone", default="未填寫")
    """
    try:
        for key in keys:
            data = data[key]
        return data
    except (KeyError, TypeError):
        return default


# 使用範例
if __name__ == "__main__":
    db_url = get_env("DATABASE_URL")
    print(format_currency(1_234_567))  # NT$1,234,567

    order_ids = list(range(1, 101))  # 100 筆訂單 ID
    for batch in chunks(order_ids, 10):
        print(f"處理批次：{batch[:3]}...（共 {len(batch)} 筆）")
```
